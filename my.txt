# Внутри контейнера
apt-get update
apt-get install -y postgresql-11

su - postgres
sudo chown -R 999:999 /var/storage/sonarqube/db15
/usr/lib/postgresql/15/bin/initdb -D /var/lib/postgresql/15/data -E UTF8 --locale=en_US.utf8


# Запустим сервер Postgres 15
/usr/lib/postgresql/15/bin/pg_ctl -D /var/lib/postgresql/15/data -l /tmp/pg15.log start

# Подождем немного
sleep 3

# Создадим пользователя sonar
/usr/lib/postgresql/15/bin/psql -d postgres -c "
CREATE USER sonar WITH 
  SUPERUSER       -- Полные права суперпользователя
  CREATEDB        -- Может создавать базы данных
  CREATEROLE      -- Может создавать роли
  INHERIT         -- Наследует права
  LOGIN           -- Может логиниться
  REPLICATION     -- Может делать репликацию
  BYPASSRLS       -- Обходит RLS (Row Level Security)
  PASSWORD 'sonar';  -- Пароль
"
